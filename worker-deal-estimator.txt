// Cloudflare Worker: Deal Estimator Endpoint
// Deploy to: https://acp-deal-estimator.owner-1ab.workers.dev

export default {
  async fetch(request, env) {
    const url = new URL(request.url);

    // Handle CORS preflight
    if (request.method === 'OPTIONS') {
      return new Response(null, {
        headers: {
          'Access-Control-Allow-Origin': 'https://acp-lending.com',
          'Access-Control-Allow-Methods': 'POST, OPTIONS',
          'Access-Control-Allow-Headers': 'Content-Type',
        },
      });
    }

    // POST /api/deal-estimator - Form submission
    if (url.pathname === '/api/deal-estimator' && request.method === 'POST') {
      try {
        const formData = await request.json();

        // Validate required fields
        const required = [
          'property_address', 'property_type', 'purchase_price', 'arv',
          'repair_budget', 'borrower_credit', 'borrower_name',
          'borrower_email', 'borrower_phone'
        ];

        for (const field of required) {
          if (!formData[field]) {
            return new Response(
              JSON.stringify({ success: false, error: `Missing ${field}` }),
              { status: 400, headers: { 'Content-Type': 'application/json' } }
            );
          }
        }

        // Send alert email to Steve
        const subject = `New Deal Estimator Submission: ${formData.property_address}`;
        const emailBody = `
New Deal Estimator Submission

Borrower: ${formData.borrower_name}
Email: ${formData.borrower_email}
Phone: ${formData.borrower_phone}

Property: ${formData.property_address}
Type: ${formData.property_type}
Purchase Price: $${parseInt(formData.purchase_price).toLocaleString()}
After-Repair Value: $${parseInt(formData.arv).toLocaleString()}
Repair Budget: $${parseInt(formData.repair_budget).toLocaleString()}
Borrower FICO: ${formData.borrower_credit}

Submitted: ${new Date(formData.timestamp).toLocaleString()}

Next Steps:
1. Review the deal information in Deal Desk
2. Generate full Deal Analysis Report
3. Send summary email + $99 report offer to borrower
4. Follow up by phone or email

Contact: ${formData.borrower_phone}
        `;

        // Send email via Gmail (using app password stored in env.GMAIL_APP_PASSWORD)
        const gmailAuth = btoa(`steve@acp-lending.com:${env.GMAIL_APP_PASSWORD}`);

        const emailResponse = await fetch('https://www.googleapis.com/gmail/v1/users/me/messages/send', {
          method: 'POST',
          headers: {
            'Authorization': `Basic ${gmailAuth}`,
            'Content-Type': 'application/json'
          },
          body: JSON.stringify({
            raw: btoa(
              `From: steve@acp-lending.com\r\n` +
              `To: steve@acp-lending.com\r\n` +
              `Subject: ${subject}\r\n` +
              `\r\n` +
              emailBody
            )
          })
        });

        // Return success response
        return new Response(
          JSON.stringify({
            success: true,
            message: 'Deal submitted to Deal Desk. Steve will contact you within 24 hours.',
            submissionId: `EST-${Date.now()}`
          }),
          {
            status: 200,
            headers: {
              'Content-Type': 'application/json',
              'Access-Control-Allow-Origin': 'https://acp-lending.com',
            }
          }
        );

      } catch (error) {
        console.error('Error processing deal estimator submission:', error);
        return new Response(
          JSON.stringify({ success: false, error: error.message }),
          { status: 500, headers: { 'Content-Type': 'application/json' } }
        );
      }
    }

    // 404 for unmatched routes
    return new Response('Not Found', { status: 404 });
  }
};
